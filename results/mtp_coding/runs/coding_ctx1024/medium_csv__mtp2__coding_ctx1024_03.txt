import csv


def read_data(filename: str) -> list[dict]:
    """Read CSV file and return list of dictionaries."""
    with open(filename, newline='', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        rows = []
        for row in reader:
            rows.append(row)
    return rows


def compute_average_score(rows: list[dict]) -> float:
    """Compute the average score from the list of rows."""
    if not rows:
        return 0.0
    total = 0.0
    count = 0
    for row in rows:
        try:
            score = float(row['score'])
            total += score
            count += 1
        except (ValueError, KeyError):
            continue
    if count == 0:
        return 0.0
    return total / count


def get_names_above_average(rows: list[dict], average: float) -> list[str]:
    """Return names of entries whose score is above the given average."""
    names = []
    for row in rows:
        try:
            score = float(row['score'])
        except (ValueError, KeyError):
            continue
        if score > average:
            name = row.get('name', '')
            if name:
                names.append(name)
    return names


def main():
    filename = 'data.csv'
    rows = read_data(filename)
    average = compute_average_score(rows)
    names = get_names_above_average(rows, average)
    for name in names:
        print(name)


if __name__ == '__main__':
    main()
