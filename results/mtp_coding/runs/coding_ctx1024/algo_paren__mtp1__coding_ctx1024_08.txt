def is_balanced(s: str) -> bool:
    """
    Determine whether the brackets in the given string are balanced.

    Args:
        s: A string containing only the characters '(', ')', '{', '}', '[', ']'.

    Returns:
        True if the brackets are balanced, False otherwise.

    Examples:
        >>> is_balanced("()")
        True
        >>> is_balanced("([{}])")
        True
        >>> is_balanced("([)]")
        False
    """
    stack = []
    matching = {')': '(', '}': '{', ']': '['}
    opening = {'(', '{', '['}

    for char in s:
        if char in opening:
            stack.append(char)
        elif char in matching:
            if not stack or stack[-1] != matching[char]:
                return False
            stack.pop()
        else:
            # Character is not a bracket, but per problem statement this shouldn't happen
            return False

    return len(stack) == 0
