import heapq
from typing import Dict, List, Tuple


def dijkstra(graph: Dict[int, List[Tuple[int, int]]], source: int) -> Dict[int, int]:
    """
    Compute shortest paths from source to all reachable nodes using Dijkstra's algorithm.
    
    Args:
        graph: A dictionary mapping each node to a list of (neighbor, weight) tuples.
        source: The starting node.
        
    Returns:
        A dictionary mapping each reachable node to its shortest distance from source.
    """
    # Initialize distances
    distances: Dict[int, int] = {source: 0}
    
    # Min-heap: (distance, node)
    heap: List[Tuple[int, int]] = [(0, source)]
    
    while heap:
        dist, node = heapq.heappop(heap)
        
        # If we've already found a shorter path, skip
        if dist > distances.get(node, float('inf')):
            continue
        
        # Explore neighbors
        if node in graph:
            for neighbor, weight in graph[node]:
                new_dist = dist + weight
                # If we found a shorter path to neighbor, update
                if new_dist < distances.get(neighbor, float('inf')):
                    distances[neighbor] = new_dist
                    heapq.heappush(heap, (new_dist, neighbor))
    
    return distances
