import bisect


def longest_increasing_subsequence_length(nums: list[int]) -> int:
    """
    Compute the length of the longest strictly increasing subsequence (LIS)
    using the O(n log n) patience-sorting method.

    The algorithm maintains a list `tails` where `tails[i]` is the smallest
    tail element for an increasing subsequence of length `i + 1`. For each
    number in the input, we use binary search to find the position where it
    should be placed:
    - If the number is greater than all elements in `tails`, it extends the
      longest subsequence found so far (append to `tails`).
    - Otherwise, it replaces the first element in `tails` that is >= to it,
      which keeps the tails as small as possible for future extensions.

    The length of `tails` at the end equals the length of the LIS.

    Args:
        nums: A list of integers.

    Returns:
        The length of the longest strictly increasing subsequence.
    """
    if not nums:
        return 0

    tails = []
    for num in nums:
        # Find the leftmost position in tails where tails[pos] >= num
        pos = bisect.bisect_left(tails, num)
        if pos == len(tails):
            tails.append(num)
        else:
            tails[pos] = num

    return len(tails)
