import bisect


def longest_increasing_subsequence_length(nums: list[int]) -> int:
    """
    Compute the length of the longest strictly increasing subsequence (LIS)
    using the O(n log n) patience-sorting method.

    The algorithm maintains a list `tails` where `tails[i]` is the smallest
    tail element for an increasing subsequence of length `i + 1`. For each
    number in the input, we use binary search to find the position where it
    can replace an existing tail (to keep tails as small as possible) or
    extend the longest subsequence found so far. The length of `tails` at the
    end equals the length of the LIS.

    Args:
        nums: A list of integers.

    Returns:
        The length of the longest strictly increasing subsequence.
    """
    if not nums:
        return 0

    tails = []  # tails[i] = smallest tail of an increasing subsequence of length i+1

    for num in nums:
        # Find the leftmost position in tails where tails[pos] >= num
        # This ensures strict increase (we replace the first element >= num)
        pos = bisect.bisect_left(tails, num)

        if pos == len(tails):
            # num is larger than all elements in tails, extend the longest subsequence
            tails.append(num)
        else:
            # Replace tails[pos] with num to keep the tail as small as possible
            tails[pos] = num

    return len(tails)
