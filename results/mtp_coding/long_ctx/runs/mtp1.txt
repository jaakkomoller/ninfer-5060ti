from __future__ import annotations

from typing import Dict, Iterable, List


def completion_ratio(done: int, total: int) -> float:
    if total <= 0:
        return 0.0
    return done / total


def tag_histogram(tasks: Iterable) -> Dict[str, int]:
    counts: Dict[str, int] = {}
    for t in tasks:
        for tag in t.tags:
            counts[tag] = counts.get(tag, 0) + 1
    return counts
