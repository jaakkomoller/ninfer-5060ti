```bash
#!/bin/bash

# Find all .log files recursively, print path and line count,
# exit non-zero if any file has more than 1000 lines.

exit_code=0

# Use find to locate all *.log files, then for each file, count lines
# and check if it exceeds 1000 lines.
while IFS= read -r -d '' file; do
    line_count=$(wc -l < "$file")
    echo "$file: $line_count"
    if [ "$line_count" -gt 1000 ]; then
        exit_code=1
    fi
done < <(find . -type f -name '*.log' -print0)

exit $exit_code
```
