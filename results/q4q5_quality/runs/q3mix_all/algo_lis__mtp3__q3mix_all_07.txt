import bisect

def longest_increasing_subsequence(nums: list[int]) -> int:
    """
    Compute the length of the longest strictly increasing subsequence (LIS)
    using the O(n log n) patience-sorting method.

    The algorithm maintains a list `tails` where `tails[i]` is the smallest
    tail element for all increasing subsequences of length `i + 1`. For each
    number in the input, we use binary search to find the position where it
    should be placed:
    - If the number is greater than all elements in `tails`, it extends the
      longest subsequence.
    - Otherwise, it replaces the first element in `tails` that is greater
      than or equal to it, keeping the tails as small as possible to allow
      for longer subsequences in the future.

    The length of `tails` at the end equals the length of the LIS.

    Args:
        nums: A list of integers.

    Returns:
        The length of the longest strictly increasing subsequence.
    """
    if not nums:
        return 0

    tails = []  # tails[i] = smallest tail for increasing subsequence of length i+1

    for num in nums:
        # Find the leftmost position where tails[pos] >= num
        # This ensures strict increase (we replace the first element >= num)
        pos = bisect.bisect_left(tails, num)

        if pos == len(tails):
            # num is greater than all current tails, extend the LIS
            tails.append(num)
        else:
            # Replace tails[pos] with num to keep tails as small as possible
            tails[pos] = num

    return len(tails)
