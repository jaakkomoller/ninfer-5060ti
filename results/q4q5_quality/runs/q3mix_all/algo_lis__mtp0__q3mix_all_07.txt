import bisect

def longest_increasing_subsequence(nums: list[int]) -> int:
    """
    Compute the length of the longest strictly increasing subsequence (LIS)
    using the O(n log n) patience-sorting method.

    The algorithm maintains a list `tails` where `tails[i]` is the smallest
    tail element for all increasing subsequences of length i+1. For each
    element in the input, we use binary search to find its position in `tails`:
    - If the element is larger than all elements in `tails`, it extends the
      longest subsequence.
    - Otherwise, it replaces the first element in `tails` that is >= to it,
      which keeps the tails as small as possible for future extensions.

    The length of `tails` at the end equals the length of the LIS.

    Args:
        nums: A list of integers.

    Returns:
        The length of the longest strictly increasing subsequence.
    """
    if not nums:
        return 0

    tails = []  # tails[i] = smallest tail of an increasing subsequence of length i+1

    for num in nums:
        # Find the position where num should be placed in tails
        # bisect_left finds the first index where tails[index] >= num
        pos = bisect.bisect_left(tails, num)

        # If pos is at the end, num extends the longest subsequence
        if pos == len(tails):
            tails.append(num)
        else:
            # Replace tails[pos] with num to keep tails as small as possible
            tails[pos] = num

    return len(tails)
