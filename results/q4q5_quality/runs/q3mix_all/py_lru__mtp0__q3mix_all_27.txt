class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}
        # Use a doubly linked list for O(1) operations
        # Node: [key, value, prev, next]
        # head and tail are dummy nodes
        self.head = [None, None, None, None]
        self.tail = [None, None, None, None]
        self.head[2] = self.tail
        self.head[3] = self.tail
        self.tail[0] = self.head
        self.tail[1] = self.head

    def _remove(self, node):
        """Remove a node from the linked list."""
        prev_node = node[2]
        next_node = node[3]
        prev_node[3] = next_node
        next_node[2] = prev_node

    def _add_to_head(self, node):
        """Add a node right after the head (most recently used)."""
        next_node = self.head[3]
        self.head[3] = node
        node[2] = self.head
        node[3] = next_node
        next_node[2] = node

    def get(self, key):
        if key not in self.cache:
            return -1
        node = self.cache[key]
        # Move to head (most recently used)
        self._remove(node)
        self._add_to_head(node)
        return node[1]

    def put(self, key, value):
        if key in self.cache:
            # Update value and move to head
            node = self.cache[key]
            node[1] = value
            self._remove(node)
            self._add_to_head(node)
        else:
            # Create new node
            node = [key, value, None, None]
            self.cache[key] = node
            self._add_to_head(node)
            # If capacity exceeded, remove least recently used
            if len(self.cache) > self.capacity:
                lru_node = self.tail[2]
                self._remove(lru_node)
                del self.cache[lru_node[0]]
