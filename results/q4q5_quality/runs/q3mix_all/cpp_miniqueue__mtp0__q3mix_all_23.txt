#include <deque>

class MiniQueue {
private:
    std::deque<int> dq;

public:
    void push(int x) {
        dq.push_back(x);
    }

    bool pop(int &x) {
        if (dq.empty()) {
            return false;
        }
        x = dq.front();
        dq.pop_front();
        return true;
    }

    int front() const {
        return dq.front();
    }

    int size() const {
        return static_cast<int>(dq.size());
    }
};
