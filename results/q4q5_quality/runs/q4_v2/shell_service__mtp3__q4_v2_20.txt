```ini
# /etc/systemd/system/myapp.service
#
# Systemd service unit file for myapp
#
# This service runs /usr/local/bin/myapp on port 8080 as the 'myapp' user,
# restarts on failure with a 5-second delay, and waits for network-online.target.

[Unit]
# Description: Human-readable description of the service
Description=My Application Service

# After: Ensures this service starts after network-online.target is reached.
# network-online.target is a passive target that is reached when the network
# is considered online (i.e., at least one network interface is up and has an IP address).
After=network-online.target

# Wants: Creates a weak dependency on network-online.target, ensuring it is
# pulled in if not already active. This is standard practice for network-dependent services.
Wants=network-online.target

# ConditionPathExists: Optional safety check to ensure the binary exists before starting.
# This prevents the service from failing to start if the binary is missing.
ConditionPathExists=/usr/local/bin/myapp

[Service]
# Type: Specifies the type of the service. 'simple' is the default and most common,
# meaning the process is considered started immediately after fork/exec.
Type=simple

# ExecStart: The command to execute when the service starts.
# Runs /usr/local/bin/myapp with the --port 8080 argument.
ExecStart=/usr/local/bin/myapp --port 8080

# User: Specifies the user account under which the service process runs.
# Running as a dedicated non-root user improves security by limiting the impact of potential vulnerabilities.
User=myapp

# Group: Specifies the group under which the service process runs.
# Using a dedicated group provides additional isolation.
Group=myapp

# Restart: Specifies the restart policy. 'on-failure' means the service will be
# restarted if it exits with a non-zero exit code, is killed by a signal, or times out.
# It will NOT restart on clean exit (exit code 0).
Restart=on-failure

# RestartSec: Specifies the delay (in seconds) before restarting the service after a failure.
# A 5-second delay prevents rapid restart loops in case of persistent failures.
RestartSec=5

# StandardOutput: Redirects standard output to the journal for logging.
StandardOutput=journal

# StandardError: Redirects standard error to the journal for logging.
StandardError=journal

# WorkingDirectory: Sets the working directory for the service process.
# Using /tmp or a dedicated directory can help with relative path resolution.
WorkingDirectory=/tmp

# Environment: Sets environment variables for the service process.
# You can add any required environment variables here.
Environment=MYAPP_ENV=production

[Install]
# WantedBy: Specifies the target that this service should be started by.
# 'multi-user.target' is the standard target for multi-user systems (non-graphical).
# This makes the service start automatically at boot.
WantedBy=multi-user.target
```
