from typing import List


def two_sum(nums: List[int], target: int) -> List[int]:
    """
    Given a list of integers and a target sum, return the indices of the two
    numbers that add up to the target.

    Uses a hash map to achieve O(n) time complexity.

    Args:
        nums: A list of integers.
        target: The target sum to find.

    Returns:
        A list containing the two indices whose corresponding values sum to target.

    Example:
        >>> two_sum([2, 7, 11, 15], 9)
        [0, 1]
    """
    seen: dict[int, int] = {}
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:
            return [seen[complement], i]
        seen[num] = i
    # The problem guarantees exactly one solution, so this line is never reached.
    return []
