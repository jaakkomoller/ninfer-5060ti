import bisect


def longest_increasing_subsequence(nums: list[int]) -> int:
    """
    Compute the length of the longest strictly increasing subsequence (LIS)
    using the patience-sorting (binary search) method in O(n log n) time.

    The algorithm maintains a list `tails` where `tails[i]` is the smallest
    tail element for an increasing subsequence of length `i + 1`. For each
    number in the input, we use binary search to find the position where it
    should be placed in `tails` (the first element >= current number for
    strictly increasing). If the number is larger than all elements in
    `tails`, it extends the longest subsequence. Otherwise, it replaces the
    first element that is >= it, which keeps `tails` as small as possible
    for each subsequence length, allowing longer subsequences to be formed
    later.

    Args:
        nums: A list of integers.

    Returns:
        The length of the longest strictly increasing subsequence.
    """
    if not nums:
        return 0

    tails = []  # tails[i] = smallest tail of an increasing subsequence of length i+1

    for num in nums:
        # Find the leftmost index in tails where tails[index] >= num
        # This is equivalent to finding the position to replace for strictly increasing
        idx = bisect.bisect_left(tails, num)

        if idx == len(tails):
            # num is greater than all elements in tails, so it extends the longest subsequence
            tails.append(num)
        else:
            # Replace the first element that is >= num with num
            # This keeps tails as small as possible for each length
            tails[idx] = num

    return len(tails)
