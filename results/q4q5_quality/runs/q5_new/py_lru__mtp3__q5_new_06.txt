class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}
        # Doubly linked list nodes: key -> [key, value, prev, next]
        # Use dummy head and tail for easier manipulation
        self.head = [None, None, None, None]  # dummy head
        self.tail = [None, None, None, None]  # dummy tail
        self.head[2] = None
        self.head[3] = self.tail
        self.tail[2] = self.head
        self.tail[3] = None

    def _remove(self, node):
        """Remove node from the linked list."""
        prev_node = node[2]
        next_node = node[3]
        prev_node[3] = next_node
        next_node[2] = prev_node

    def _add_to_front(self, node):
        """Add node right after head (most recently used)."""
        next_node = self.head[3]
        self.head[3] = node
        node[2] = self.head
        node[3] = next_node
        next_node[2] = node

    def _move_to_front(self, node):
        """Move node to front (mark as most recently used)."""
        self._remove(node)
        self._add_to_front(node)

    def get(self, key):
        if key not in self.cache:
            return -1
        node = self.cache[key]
        self._move_to_front(node)
        return node[1]

    def put(self, key, value):
        if key in self.cache:
            node = self.cache[key]
            node[1] = value
            self._move_to_front(node)
        else:
            if len(self.cache) >= self.capacity:
                # Evict least recently used (node before tail)
                lru_node = self.tail[2]
                self._remove(lru_node)
                del self.cache[lru_node[0]]
            # Add new node
            new_node = [key, value, None, None]
            self.cache[key] = new_node
            self._add_to_front(new_node)
