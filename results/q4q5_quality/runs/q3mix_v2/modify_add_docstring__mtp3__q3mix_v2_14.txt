def clamp(value, low, high):
    """Clamp a value between a lower and upper bound.

    Args:
        value: The value to be clamped.
        low: The lower bound.
        high: The upper bound.

    Returns:
        The value clamped between low and high.

    Example:
        >>> clamp(15, 0, 10)
        10
    """
    return max(low, min(high, value))
