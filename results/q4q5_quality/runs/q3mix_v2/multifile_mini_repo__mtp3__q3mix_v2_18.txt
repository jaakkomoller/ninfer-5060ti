```python
class Counter:
    def __init__(self):
        self._data = {}
    def incr(self, key, by=1):
        self._data[key] = self._data.get(key, 0) + by
    def get(self, key):
        return self._data.get(key, 0)
    def __str__(self):
        items = sorted(self._data.items(), key=lambda x: (-x[1], x[0]))
        return ', '.join(f'{k}:{v}' for k, v in items)
```
