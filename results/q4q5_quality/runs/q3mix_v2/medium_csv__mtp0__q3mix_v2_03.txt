```python
import csv


def main():
    scores = []
    entries = []

    with open('data.csv', newline='') as f:
        reader = csv.DictReader(f)
        for row in reader:
            try:
                score = float(row['score'])
            except (ValueError, KeyError):
                continue
            entries.append((row['name'], score))
            scores.append(score)

    if not scores:
        print("No valid entries found.")
        return

    average = sum(scores) / len(scores)

    above_average = [name for name, score in entries if score > average]

    if above_average:
        for name in above_average:
            print(name)
    else:
        print("No entries above average.")


if __name__ == '__main__':
    main()
```
