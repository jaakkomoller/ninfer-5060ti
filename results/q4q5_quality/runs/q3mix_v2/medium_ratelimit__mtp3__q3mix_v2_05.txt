import time


class RateLimiter:
    def __init__(self, rate_per_sec: float, burst: int):
        """
        Initialize the RateLimiter.
        
        Args:
            rate_per_sec: The rate at which tokens are added per second.
            burst: The maximum number of tokens that can be accumulated (bucket capacity).
        """
        self.rate_per_sec = rate_per_sec
        self.burst = burst
        self.tokens = float(burst)  # Start with a full bucket
        self.last_time = time.monotonic()

    def acquire(self) -> bool:
        """
        Try to acquire a token. Returns True if a token is available, False otherwise.
        """
        now = time.monotonic()
        elapsed = now - self.last_time
        self.last_time = now

        # Add tokens based on elapsed time
        self.tokens += elapsed * self.rate_per_sec

        # Cap tokens at burst
        if self.tokens > self.burst:
            self.tokens = float(self.burst)

        # Try to consume a token
        if self.tokens >= 1.0:
            self.tokens -= 1.0
            return True
        else:
            return False
