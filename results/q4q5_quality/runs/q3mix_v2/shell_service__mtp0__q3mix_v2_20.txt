# /etc/systemd/system/myapp.service
# Systemd service unit file for myapp

[Unit]
# Description shown in systemctl status output
Description=My Application Service

# Ensure the service starts only after the network is fully online
# This prevents the app from starting before network interfaces are up
After=network-online.target

# Create a dependency on network-online.target so systemd waits for it
Wants=network-online.target

[Service]
# Run the process as the unprivileged user 'myapp'
User=myapp

# The full command to execute, including the port argument
ExecStart=/usr/local/bin/myapp --port 8080

# Restart the service if it exits with a non-zero status (failure)
Restart=on-failure

# Wait 5 seconds before restarting a failed service
RestartSec=5

# Optional: set working directory if needed
# WorkingDirectory=/var/lib/myapp

[Install]
# Specify the target to enable the service at boot
# This allows 'systemctl enable myapp' to work correctly
WantedBy=multi-user.target
