import heapq
from typing import Dict, List, Tuple


def dijkstra(graph: Dict[int, List[Tuple[int, int]]], source: int) -> Dict[int, float]:
    """
    Compute shortest paths from source to all reachable nodes using Dijkstra's algorithm.
    
    Args:
        graph: A dictionary mapping each node to a list of (neighbor, weight) tuples.
        source: The starting node.
    
    Returns:
        A dictionary mapping each reachable node to its shortest distance from source.
    """
    # Initialize distances to infinity for all nodes
    # We'll collect all nodes from the graph keys and their neighbors
    all_nodes = set(graph.keys())
    for node in graph:
        for neighbor, weight in graph[node]:
            all_nodes.add(neighbor)
    
    # Initialize distances
    distances: Dict[int, float] = {node: float('inf') for node in all_nodes}
    distances[source] = 0
    
    # Priority queue: (distance, node)
    pq: List[Tuple[float, int]] = [(0, source)]
    
    # Process nodes
    while pq:
        dist, node = heapq.heappop(pq)
        
        # If we've already found a shorter path, skip
        if dist > distances[node]:
            continue
        
        # Explore neighbors
        for neighbor, weight in graph.get(node, []):
            new_dist = dist + weight
            if new_dist < distances[neighbor]:
                distances[neighbor] = new_dist
                heapq.heappush(pq, (new_dist, neighbor))
    
    # Remove unreachable nodes (distance is still infinity)
    result = {node: dist for node, dist in distances.items() if dist != float('inf')}
    
    return result
