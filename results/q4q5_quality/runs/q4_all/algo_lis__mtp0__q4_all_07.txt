import bisect


def longest_increasing_subsequence_length(nums: list[int]) -> int:
    """
    Compute the length of the longest strictly increasing subsequence (LIS)
    using the patience-sorting (binary search) method in O(n log n) time.

    The algorithm maintains a list `tails` where `tails[i]` is the smallest
    tail element for an increasing subsequence of length `i + 1`. For each
    number in the input, we use binary search to find the position where it
    should be placed:
    - If the number is larger than all elements in `tails`, it extends the
      longest subsequence.
    - Otherwise, it replaces the first element in `tails` that is greater than
      or equal to it, keeping `tails` sorted and allowing future elements to
      potentially extend subsequences.

    The length of `tails` at the end equals the length of the LIS.

    Args:
        nums: A list of integers.

    Returns:
        The length of the longest strictly increasing subsequence.
    """
    if not nums:
        return 0

    tails = []  # tails[i] = smallest tail of an increasing subsequence of length i+1

    for num in nums:
        # Find the leftmost position where num can be placed
        # bisect_left finds the first index where num can be inserted to keep order
        pos = bisect.bisect_left(tails, num)

        if pos == len(tails):
            # num is greater than all elements in tails, extend the longest subsequence
            tails.append(num)
        else:
            # Replace the first element >= num to maintain the smallest possible tail
            tails[pos] = num

    return len(tails)
