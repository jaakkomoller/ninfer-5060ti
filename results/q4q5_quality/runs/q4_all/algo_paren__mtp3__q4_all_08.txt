def is_balanced(s: str) -> bool:
    """
    Determine whether the brackets in a string are balanced.

    A string is considered balanced if:
    - Every opening bracket has a corresponding closing bracket of the same type.
    - Brackets are properly nested (i.e., they close in the reverse order they were opened).

    Args:
        s: A string containing only the characters '(', ')', '{', '}', '[', ']'.

    Returns:
        True if the brackets are balanced, False otherwise.

    Examples:
        >>> is_balanced("()")
        True
        >>> is_balanced("([{}])")
        True
        >>> is_balanced("([)]")
        False
    """
    stack = []
    matching = {
        ')': '(',
        '}': '{',
        ']': '['
    }
    opening = {'(', '{', '['}

    for char in s:
        if char in opening:
            stack.append(char)
        elif char in matching:
            if not stack or stack[-1] != matching[char]:
                return False
            stack.pop()
        else:
            # Character is not a bracket, but per problem statement, string contains only brackets
            # So this case shouldn't occur, but we can ignore or return False
            return False

    return len(stack) == 0
