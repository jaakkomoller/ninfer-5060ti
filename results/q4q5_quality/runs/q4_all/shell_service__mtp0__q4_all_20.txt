```ini
# /etc/systemd/system/myapp.service
#
# Systemd service unit file for myapp
#
# This service runs /usr/local/bin/myapp on port 8080 as the 'myapp' user,
# restarts on failure with a 5-second delay, and starts after network-online.target.

[Unit]
# Description: Human-readable description of the service, shown in systemctl status
Description=My Application Service

# After: Specifies that this service should be started after network-online.target is reached.
# This ensures the network is fully up before the application starts, which is important
# for services that depend on network connectivity (e.g., binding to a port or making outbound connections).
After=network-online.target

# Wants: Creates a weak dependency on network-online.target, ensuring it is pulled in
# if not already active. This is a best practice for network-dependent services.
Wants=network-online.target

[Service]
# Type: Specifies the type of the service. 'simple' is the default and most common type,
# where the service is considered started as soon as the main process is forked.
Type=simple

# User: Runs the service as the unprivileged user 'myapp' instead of root.
# This follows the principle of least privilege for security.
User=myapp

# ExecStart: The command to execute when the service starts.
# This runs the myapp binary with the --port 8080 argument.
ExecStart=/usr/local/bin/myapp --port 8080

# Restart: Specifies the condition under which the service should be restarted.
# 'on-failure' means the service will be restarted if it exits with a non-zero exit code,
# is killed by a signal, or fails to start.
Restart=on-failure

# RestartSec: The delay in seconds before restarting the service after a failure.
# A 5-second delay prevents rapid restart loops if the service is crashing repeatedly.
RestartSec=5

# WorkingDirectory: Sets the working directory for the service process.
# This is optional but good practice to ensure relative paths in the application work as expected.
WorkingDirectory=/usr/local

[Install]
# WantedBy: Specifies the target that this service should be started by.
# 'multi-user.target' is the standard target for normal system operation (non-graphical).
WantedBy=multi-user.target
```
