import csv


def read_data(filename: str) -> list[dict[str, str]]:
    """Read a CSV file and return a list of dictionaries."""
    with open(filename, newline='', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        return list(reader)


def compute_average_score(data: list[dict[str, str]]) -> float:
    """Compute the average score from the data."""
    if not data:
        return 0.0
    total = sum(float(row['score']) for row in data)
    return total / len(data)


def get_names_above_average(data: list[dict[str, str]], average: float) -> list[str]:
    """Return names of entries whose score is above the average."""
    return [row['name'] for row in data if float(row['score']) > average]


def main():
    data = read_data('data.csv')
    if not data:
        print("No data found.")
        return
    average = compute_average_score(data)
    names = get_names_above_average(data, average)
    for name in names:
        print(name)


if __name__ == '__main__':
    main()
